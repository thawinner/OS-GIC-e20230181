# Task 2: Directory Listing

./dir_list_lib

gcc -o dir_list_sys dir_list_sys.c
./dir_list_sys

# Check that both versions give identical output

./dir_list_sys > /tmp/dir_sys.txt
diff /tmp/dir_lib.txt /tmp/dir_sys.txt && echo "IDENTICAL"
